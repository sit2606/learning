#include <iostream>
#include <string>
int input()
{
    int x;
    std::cout << "Type down integer positive number to derive it's square root" << "\n";
    std::cin >> x;
    return x;
}
double calc(double target)
{
    double x = 1;
    double oldx;
    do {
        oldx = x;
        x = (x + target / x) / 2;
    }
    while (oldx != x);
    return(x);
}
int main()
{
    int number = input();
    double result = calc(number);
    std::cout << "result is " << result << "\n";
    std::cout << "your number was " << result*result << "\n";
}
